```ts
import type { NextConfig } from 'next'

/**
 * Sent on every response. There is no Content-Security-Policy: Next needs a
 * per-request nonce for its inline scripts, and reading one makes every page
 * render on demand, which would undo prerendering the phone pages.
 */
const securityHeaders = [
  { key: 'X-Content-Type-Options', value: 'nosniff' },
  // No one has a reason to put the site in a frame; this stops clickjacking.
  { key: 'X-Frame-Options', value: 'DENY' },
  // Shops still see visits came from us, but never which page or search.
  { key: 'Referrer-Policy', value: 'strict-origin-when-cross-origin' },
  { key: 'Permissions-Policy', value: 'camera=(), microphone=(), geolocation=(), browsing-topics=()' },
]

const nextConfig: NextConfig = {
  async headers() {
    return [{ source: '/:path*', headers: securityHeaders }]
  },
}

export default nextConfig
```

```ts
import { BRANDS, slug } from '@/normalize/phone'

/**
 * Site identity, used by metadata, canonical URLs, the sitemap and JSON-LD.
 *
 * The absolute origin has to be known at build time, because canonical URLs
 * and sitemap entries cannot be relative. On Vercel the production domain is
 * supplied automatically; set NEXT_PUBLIC_SITE_URL once mobileprices.lk points
 * at it (phase 4), or every canonical keeps the vercel.app host.
 */
export const SITE_URL = (
  process.env.NEXT_PUBLIC_SITE_URL ||
  (process.env.VERCEL_PROJECT_PRODUCTION_URL
    ? `https://${process.env.VERCEL_PROJECT_PRODUCTION_URL}`
    : 'http://localhost:3000')
).replace(/\/$/, '')

export const SITE_NAME = 'MobilePrices.lk'

export const canonical = (path: string) => `${SITE_URL}${path}`

/**
 * The six launch shops, in the canvas order. Keep in step with
 * `src/scrapers/index.ts`: the footer tells visitors these are the shops we
 * cover.
 */
export const TRACKED_SHOPS = [
  'lifemobile.lk',
  'celltronics.lk',
  'geniusmobile.lk',
  'banana.lk',
  'idealz.lk',
  'doctormobile.lk',
] as const

/** A brand is also its URL: /samsung, /oneplus, /oppo. */
export const brandSlug = (brand: string) => slug(brand)

const BY_SLUG = new Map(BRANDS.map((b) => [slug(b), b]))

/** Every in-scope brand's slug, in the canonical brand order. */
export const BRAND_SLUGS: string[] = [...BY_SLUG.keys()]

export const brandFromSlug = (value: string): string | null => BY_SLUG.get(value) ?? null

/**
 * Pale tint and darker ink per brand, for tiles and the brand page band. The
 * first eight are the canvas's; the canvas has none for the other eight, which
 * reuse them in the same order. Brands are text, never logos (decisions,
 * 2026-10-23).
 */
const TINTS: [tint: string, ink: string][] = [
  ['#EEF1F5', '#1A1F2B'], // Apple
  ['#E9EEFF', '#3B55C9'], // Samsung
  ['#FDEDE1', '#A4511F'], // Xiaomi
  ['#F5E9FB', '#8A3BA8'], // Honor
  ['#FDE9E6', '#B23A30'], // Huawei
  ['#E3F5EC', '#1E7A55'], // OPPO
  ['#E1F3F8', '#1D6F86'], // vivo
  ['#FCF0DC', '#8C5E1A'], // realme
]

export function brandStyle(brand: string): { tint: string; ink: string } {
  const [tint, ink] = TINTS[Math.max(0, BRANDS.indexOf(brand)) % TINTS.length]
  return { tint, ink }
}

/** "Samsung Galaxy S26 Ultra": the name people search for. */
export const phoneName = (p: { brand: string; model: string }) => `${p.brand} ${p.model}`
```

```ts
// src/app/robots.ts
import type { MetadataRoute } from 'next'

import { canonical } from '@/lib/site'

/**
 * Every page is crawlable. The main job of this file is to point at the
 * sitemap, which is how the phone pages get discovered in reasonable time.
 *
 * `/api` is disallowed to keep it out of search results, not to protect it — a
 * robots file is a request. What protects the revalidate handler is its token.
 */
export default function robots(): MetadataRoute.Robots {
  return {
    rules: { userAgent: '*', allow: '/', disallow: ['/api'] },
    sitemap: canonical('/sitemap.xml'),
    host: canonical('/').replace(/\/$/, ''),
  }
}
```

```ts
// src/app/api/revalidate/route.ts
import { revalidateTag } from 'next/cache'
import { timingSafeEqual } from 'node:crypto'

/**
 * Lets the nightly pipeline clear the read caches when it finishes. Every
 * cached read (one entry per brand) shares the tag `catalog` with a 30-minute
 * revalidate; without this, fresh prices sit behind a stale cache for up to
 * half an hour after each run. Wiring the calls (REVALIDATE_SECRET, SITE_URL)
 * is phase 4.
 */

function authorized(request: Request): boolean {
  const secret = process.env.REVALIDATE_SECRET
  if (!secret) return false

  const header = request.headers.get('authorization') ?? ''
  const token = header.startsWith('Bearer ') ? header.slice(7) : ''
  const a = Buffer.from(token)
  const b = Buffer.from(secret)
  return a.length === b.length && timingSafeEqual(a, b)
}

export async function POST(request: Request) {
  if (!authorized(request)) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 })
  }

  // Two-argument form: the single-argument `revalidateTag(tag)` is deprecated in Next 16.
  revalidateTag('catalog', 'max')

  return Response.json({ revalidated: 'catalog' })
}
```

```ts
import type { Warranty } from '@/db/schema'

/**
 * The price rules, in one pure module the pages, the share images and the
 * tests all read (spec: "Price rules").
 *
 * - Cheapest overall: any warranty class, `unstated` included.
 * - With company warranty: `company` or `apple` (Apple's own counts).
 * - A listing not seen for more than STALE_AFTER_DAYS is shown "last seen" and
 *   never holds a headline, as on PCPrice.lk.
 * - In stock beats out of stock for a headline; out of stock holds one only
 *   when nothing current is in stock.
 */

/** How old a listing may be before its price is presented as last-seen. */
export const STALE_AFTER_DAYS = 7

export type Offer = {
  shop: string
  warranty: Warranty
  /** The shop's own warranty wording, or null when it states none. */
  warrantyText: string | null
  priceLkr: number
  url: string
  inStock: boolean
  /** ISO timestamp of the scrape that saw this price. */
  seenAt: string
  /** Sri Lanka date of `seenAt`, YYYY-MM-DD. */
  seenOn: string
  stale: boolean
}

/** A `listings` row as Drizzle returns it (`numeric` comes back as a string). */
export type ListingRow = {
  shop: string
  warranty: Warranty
  warrantyText: string | null
  priceLkr: string | number
  url: string
  inStock: boolean
  scrapedAt: Date | string
}

export type PhoneRow = {
  phoneId: string
  brand: string
  model: string
  ramGb: number | null
  storageGb: number
  esimOnly: boolean
}

/** A phone configuration with its offers and headline prices, as every page reads it. */
export type PhoneSummary = PhoneRow & {
  brandSlug: string
  /** In `sortOffers` order. */
  offers: Offer[]
  overall: Offer | null
  company: Offer | null
  /** Distinct shops listing it. */
  shops: number
  /** Distinct shops with a current, in-stock price. */
  inStockShops: number
  /** Newest `seenOn`, for the sitemap. */
  updatedOn: string
  /** Newest `seenAt`, for "Checked 25 Oct at 03:00". */
  checkedAt: string
}

export type Headlines = { overall: Offer | null; company: Offer | null }

/** One day of the 90-day chart: lowest in-stock price, any warranty and company warranty. */
export type HistoryPoint = { day: string; anyLkr: number; companyLkr: number | null }

/** One day's lowest in-stock price, any warranty: what price drops are judged on. */
export type PricePoint = { day: string; lowestLkr: number }

const colomboDate = new Intl.DateTimeFormat('en-CA', {
  timeZone: 'Asia/Colombo',
  year: 'numeric',
  month: '2-digit',
  day: '2-digit',
})
const colomboDay = new Intl.DateTimeFormat('en-GB', { timeZone: 'Asia/Colombo', day: 'numeric', month: 'short' })
const colomboTime = new Intl.DateTimeFormat('en-GB', {
  timeZone: 'Asia/Colombo',
  hour: '2-digit',
  minute: '2-digit',
  hourCycle: 'h23',
})

/** The calendar day in Sri Lanka, which is what `price_history.recorded_on` holds. */
export const sriLankaDate = (d: Date) => colomboDate.format(d)

/** "25 Oct at 03:00", Sri Lanka time. */
export const checkedLabel = (iso: string) => `${colomboDay.format(new Date(iso))} at ${colomboTime.format(new Date(iso))}`

export function toOffer(row: ListingRow, now: Date): Offer {
  const seen = row.scrapedAt instanceof Date ? row.scrapedAt : new Date(row.scrapedAt)
  return {
    shop: row.shop,
    warranty: row.warranty,
    warrantyText: row.warrantyText?.trim() || null,
    priceLkr: Number(row.priceLkr),
    url: row.url,
    inStock: row.inStock,
    seenAt: seen.toISOString(),
    seenOn: sriLankaDate(seen),
    stale: seen.getTime() < now.getTime() - STALE_AFTER_DAYS * 86_400_000,
  }
}

export const isCompany = (w: Warranty) => w === 'company' || w === 'apple'

const rank = (o: Offer) => (o.stale ? 2 : o.inStock ? 0 : 1)

/** In stock, then out of stock, then last seen; cheapest first within each. */
export function sortOffers(offers: Offer[]): Offer[] {
  return [...offers].sort((a, b) => rank(a) - rank(b) || a.priceLkr - b.priceLkr)
}

export function headlines(offers: Offer[]): Headlines {
  const current = sortOffers(offers.filter((o) => !o.stale))
  return { overall: current[0] ?? null, company: current.find((o) => isCompany(o.warranty)) ?? null }
}

/** Every current price, out of stock included ("6 shops · Rs 339,900 to Rs 389,900"). */
export function priceRange(offers: Offer[]): { low: number; high: number } | null {
  const prices = offers.filter((o) => !o.stale).map((o) => o.priceLkr)
  return prices.length ? { low: Math.min(...prices), high: Math.max(...prices) } : null
}

/** `offers` must not be empty: a phone with no listing has no page. */
export function summarisePhone(phone: PhoneRow, brandSlug: string, offers: Offer[]): PhoneSummary {
  const sorted = sortOffers(offers)
  const { overall, company } = headlines(sorted)
  const newest = sorted.reduce((a, b) => (a.seenAt >= b.seenAt ? a : b))
  return {
    phoneId: phone.phoneId,
    brand: phone.brand,
    model: phone.model,
    ramGb: phone.ramGb,
    storageGb: phone.storageGb,
    esimOnly: phone.esimOnly,
    brandSlug,
    offers: sorted,
    overall,
    company,
    shops: new Set(sorted.map((o) => o.shop)).size,
    inStockShops: new Set(sorted.filter((o) => o.inStock && !o.stale).map((o) => o.shop)).size,
    updatedOn: newest.seenOn,
    checkedAt: newest.seenAt,
  }
}

export const sizeLabel = (gb: number) => (gb >= 1024 ? `${gb / 1024} TB` : `${gb} GB`)

/** "12 GB · 256 GB", "16 GB · 1 TB", Apple "256 GB" or "256 GB · eSIM". */
export function configLabel(p: { ramGb: number | null; storageGb: number; esimOnly: boolean }): string {
  return [p.ramGb !== null ? `${p.ramGb} GB` : null, sizeLabel(p.storageGb), p.esimOnly ? 'eSIM' : null]
    .filter(Boolean)
    .join(' · ')
}

export const rupees = (n: number) => `Rs ${Math.round(n).toLocaleString('en-LK')}`

/** The badge on each price. Apple's own warranty is named, in the company colour. */
export const WARRANTY_BADGE: Record<Warranty, string> = {
  company: 'Company',
  apple: 'Apple warranty',
  shop: 'Shop',
  unstated: 'Not stated',
}

/** The same, in running text: "doctormobile.lk · shop warranty". */
export const WARRANTY_PHRASE: Record<Warranty, string> = {
  company: 'company warranty',
  apple: 'Apple warranty',
  shop: 'shop warranty',
  unstated: 'warranty not stated',
}
```

```ts
import { unstable_cache } from 'next/cache'
import { and, asc, eq, inArray, sql } from 'drizzle-orm'

import { getDb } from '@/db'
import { listings, phones, priceHistory } from '@/db/schema'
import { summarisePhone, toOffer, type HistoryPoint, type Offer, type PhoneSummary } from '@/lib/prices'
import { BRAND_SLUGS, brandFromSlug } from '@/lib/site'

/**
 * The catalogue as the pages read it.
 *
 * Cached one entry per brand from day one. A single entry for every phone grows
 * toward the 2 MB Next.js allows per entry, and over the limit the write fails
 * silently and the old entry keeps being served: the bug that froze PCPrice.lk's
 * prices. Every entry shares the tag `catalog`, which /api/revalidate clears
 * after the nightly run; otherwise entries expire after 30 minutes.
 *
 * Each `read*` is the uncached query; each `load*` is its cached twin.
 */

const CACHE = { revalidate: 1800, tags: ['catalog'] }

export type Directory = Record<string, PhoneSummary[]>

export async function readBrand(brandSlug: string): Promise<PhoneSummary[]> {
  const brand = brandFromSlug(brandSlug)
  if (!brand) return []

  const db = getDb()
  const ofBrand = db.select({ phoneId: phones.phoneId }).from(phones).where(eq(phones.brand, brand))
  const [rows, offerRows] = await Promise.all([
    db.select().from(phones).where(eq(phones.brand, brand)),
    db.select().from(listings).where(inArray(listings.phoneId, ofBrand)),
  ])

  const now = new Date()
  const byPhone = new Map<string, Offer[]>()
  for (const l of offerRows) {
    const list = byPhone.get(l.phoneId)
    if (list) list.push(toOffer(l, now))
    else byPhone.set(l.phoneId, [toOffer(l, now)])
  }

  // A phone with no listing has no page.
  const out = rows.flatMap((p) => {
    const offers = byPhone.get(p.phoneId)
    return offers?.length ? [summarisePhone(p, brandSlug, offers)] : []
  })

  // Buyable first, then cheapest: the question is where to buy.
  const price = (p: PhoneSummary) => p.overall?.priceLkr ?? Number.MAX_SAFE_INTEGER
  return out.sort((a, b) => Number(b.inStockShops > 0) - Number(a.inStockShops > 0) || price(a) - price(b))
}

export const loadBrand = unstable_cache(readBrand, ['brand-phones'], CACHE)

export async function loadDirectory(): Promise<Directory> {
  const lists = await Promise.all(BRAND_SLUGS.map((s) => loadBrand(s)))
  return Object.fromEntries(BRAND_SLUGS.map((s, i) => [s, lists[i]]))
}

export async function findPhone(brandSlug: string, phoneId: string): Promise<PhoneSummary | null> {
  if (!brandFromSlug(brandSlug)) return null
  return (await loadBrand(brandSlug)).find((p) => p.phoneId === phoneId) ?? null
}

/**
 * 90 days of daily lows for one phone: any warranty, and company warranty
 * (company or Apple). In-stock prices only, so the chart agrees with the
 * headline prices above it.
 *
 * Cached per phone, not per brand: 90 days for every phone of a big brand
 * would approach the 2 MB entry limit, and each page needs only its own.
 */
export async function readPhoneHistory(phoneId: string): Promise<HistoryPoint[]> {
  const rows = await getDb()
    .select({
      day: priceHistory.recordedOn,
      any: sql<string>`min(${priceHistory.priceLkr})`,
      company: sql<string | null>`min(${priceHistory.priceLkr}) filter (where ${priceHistory.warranty} in ('company', 'apple'))`,
    })
    .from(priceHistory)
    .where(
      and(
        eq(priceHistory.phoneId, phoneId),
        eq(priceHistory.inStock, true),
        sql`${priceHistory.recordedOn} >= current_date - 90`,
      ),
    )
    .groupBy(priceHistory.recordedOn)
    .orderBy(asc(priceHistory.recordedOn))

  return rows.map((r) => ({
    day: String(r.day),
    anyLkr: Number(r.any),
    companyLkr: r.company === null ? null : Number(r.company),
  }))
}

export const loadPhoneHistory = unstable_cache(readPhoneHistory, ['phone-history'], CACHE)
```

```tsx
 * Two drawings of the same data, as on the canvas: labelled axes from 640px,

const dayFormat = new Intl.DateTimeFormat('en-US', { month: 'short', day: 'numeric', timeZone: 'UTC' })
const dayLabel = (iso: string) => dayFormat.format(new Date(`${iso}T00:00:00Z`))
const thousands = (n: number) => `${Math.round(n / 1000)}k`

type Box = { width: number; left: number; top: number; bottom: number }
const WIDE: Box = { width: 640, left: 56, top: 20, bottom: 160 }
const NARROW: Box = { width: 320, left: 0, top: 10, bottom: 110 }

function plot(points: HistoryPoint[], box: Box) {
  const values = points.flatMap((p) => (p.companyLkr === null ? [p.anyLkr] : [p.anyLkr, p.companyLkr]))
  const min = Math.min(...values)
  const max = Math.max(...values)
  const span = max - min || 1
  const x = (i: number) => box.left + (i * (box.width - box.left)) / (points.length - 1)
  const y = (v: number) => box.top + (1 - (v - min) / span) * (box.bottom - box.top)
  const line = (pick: (p: HistoryPoint) => number | null) =>
    points
      .flatMap((p, i) => {
        const v = pick(p)
        return v === null ? [] : [`${x(i).toFixed(1)},${y(v).toFixed(1)}`]
      })
      .join(' ')
  const last = points.length - 1
  return {
    min,
    max,
    any: line((p) => p.anyLkr),
    company: line((p) => p.companyLkr),
    end: { x: x(last), y: y(points[last].anyLkr) },
  }
}
```

```tsx
'use client'

import Link from 'next/link'
import { useMemo, useState } from 'react'

import type { Warranty } from '@/db/schema'
import { isCompany, rupees, sizeLabel } from '@/lib/prices'

/**
 * The filterable list on a brand page, adapted from PCPrice's CategoryList.
 *
 * Filtering runs in the browser so the page stays prerendered: the first HTML
 * holds every phone, out of stock included. A brand page is the crawl path to
 * its phone pages, so a filter that dropped rows on the server would orphan
 * them.
 *
 * Unticking a warranty class or a shop re-prices every card from the offers
 * still allowed: the question here is "where is it cheapest, on my terms".
 */

export type BrandRow = {
  phoneId: string
  href: string
  model: string
  ramGb: number | null
  storageGb: number
  esimOnly: boolean
  offers: { shop: string; warranty: Warranty; priceLkr: number; buyable: boolean }[]
}

type WarrantyFilter = 'company' | 'shop' | 'unstated'
type Sort = 'low' | 'high' | 'shops'

const WARRANTY_FILTERS: { key: WarrantyFilter; label: string; dot: string }[] = [
  { key: 'company', label: 'Company', dot: 'bg-accent' },
  { key: 'shop', label: 'Shop', dot: 'bg-shop-dot' },
  { key: 'unstated', label: 'Not stated', dot: 'shadow-[inset_0_0_0_2px_#69707F]' },
]

/** Apple's own warranty files under Company (decisions, 2026-10-25). */
const filterOf = (w: Warranty): WarrantyFilter => (w === 'company' || w === 'apple' ? 'company' : w)

function toggle<T>(set: Set<T>, value: T): Set<T> {
  const next = new Set(set)
  if (next.has(value)) next.delete(value)
  else next.add(value)
  return next
}

const legend = 'mb-3 p-0 text-[13px] font-bold uppercase tracking-[0.06em] text-ink-3'
const checkRow = 'flex min-h-10 cursor-pointer items-center gap-2.5 text-[15px]'
const checkbox = 'h-[18px] w-[18px] accent-ink'

function Chip({ on, label, onChange }: { on: boolean; label: string; onChange: () => void }) {
  return (
    <label
      className={`inline-flex h-10 cursor-pointer items-center rounded-full px-3.5 text-[14px] font-semibold has-[:focus-visible]:outline has-[:focus-visible]:outline-2 has-[:focus-visible]:outline-accent ${
        on ? 'bg-ink text-white' : 'chip edge-strong'
      }`}
    >
      <input type="checkbox" checked={on} onChange={onChange} className="sr-only" />
      {label}
    </label>
  )
}

export function BrandList({ rows, shops }: { rows: BrandRow[]; shops: string[] }) {
  const [warranties, setWarranties] = useState<Set<WarrantyFilter>>(new Set(['company', 'shop', 'unstated']))
  const [storage, setStorage] = useState<Set<number>>(new Set())
  const [ram, setRam] = useState<Set<number>>(new Set())
  const [shopsOn, setShopsOn] = useState<Set<string>>(new Set(shops))
  const [inStockOnly, setInStockOnly] = useState(false)
  const [sort, setSort] = useState<Sort>('low')
  const [open, setOpen] = useState(false)

  const storages = useMemo(() => [...new Set(rows.map((r) => r.storageGb))].sort((a, b) => a - b), [rows])
  const rams = useMemo(
    () => [...new Set(rows.flatMap((r) => (r.ramGb === null ? [] : [r.ramGb])))].sort((a, b) => a - b),
    [rows],
  )

  const shown = useMemo(() => {
    const priced = rows.flatMap((row) => {
      if (storage.size && !storage.has(row.storageGb)) return []
      if (ram.size && (row.ramGb === null || !ram.has(row.ramGb))) return []
      const offers = row.offers.filter((o) => shopsOn.has(o.shop) && warranties.has(filterOf(o.warranty)))
      if (offers.length === 0) return []
      const buyable = offers.filter((o) => o.buyable)
      if (inStockOnly && buyable.length === 0) return []
      const company = buyable.filter((o) => isCompany(o.warranty)).map((o) => o.priceLkr)
      return [
        {
          row,
          best: buyable.length ? Math.min(...buyable.map((o) => o.priceLkr)) : null,
          lastListed: Math.min(...offers.map((o) => o.priceLkr)),
          company: company.length ? Math.min(...company) : null,
          shops: new Set(offers.map((o) => o.shop)).size,
          inStock: buyable.length > 0,
        },
      ]
    })
    const price = (p: (typeof priced)[number]) => p.best ?? p.lastListed
    return priced.sort(
      (a, b) =>
        Number(b.inStock) - Number(a.inStock) ||
        (sort === 'shops' ? b.shops - a.shops : 0) ||
        (sort === 'high' ? price(b) - price(a) : price(a) - price(b)),
    )
  }, [rows, storage, ram, shopsOn, warranties, inStockOnly, sort])

  const summary =
    (storage.size ? ` with ${[...storage].sort((a, b) => a - b).map(sizeLabel).join(' or ')}` : '') +
    (inStockOnly ? ', in stock' : '')

  const reset = () => {
    setWarranties(new Set(['company', 'shop', 'unstated']))
    setStorage(new Set())
    setRam(new Set())
    setShopsOn(new Set(shops))
    setInStockOnly(false)
  }

  return (
    <div className="flex flex-wrap items-start gap-6 lg:gap-10">
      <button
        type="button"
        aria-expanded={open}
        aria-controls="filters"
        onClick={() => setOpen((o) => !o)}
        className="chip edge-strong inline-flex h-11 cursor-pointer items-center rounded-full px-4 text-[15px] font-semibold lg:hidden"
      >
        {open ? 'Hide filters' : 'Filters'}
      </button>

      <aside id="filters" aria-label="Filters" className={`${open ? 'flex' : 'hidden'} min-w-0 flex-[1_1_240px] flex-col gap-7 lg:flex`}>
        <fieldset className="m-0 border-0 p-0">
          <legend className={legend}>Warranty</legend>
          <div className="flex flex-col gap-1">
            {WARRANTY_FILTERS.map((f) => (
              <label key={f.key} className={checkRow}>
                <input
                  type="checkbox"
                  checked={warranties.has(f.key)}
                  onChange={() => setWarranties((s) => toggle(s, f.key))}
                  className={checkbox}
                />
                <span aria-hidden className={`h-2.5 w-2.5 rounded-full ${f.dot}`} />
                {f.label}
              </label>
            ))}
          </div>
        </fieldset>

        <fieldset className="m-0 border-0 p-0">
          <legend className={legend}>Storage</legend>
          <div className="flex flex-wrap gap-2">
            {storages.map((gb) => (
              <Chip key={gb} on={storage.has(gb)} label={sizeLabel(gb)} onChange={() => setStorage((s) => toggle(s, gb))} />
            ))}
          </div>
        </fieldset>

        {rams.length > 0 && (
          <fieldset className="m-0 border-0 p-0">
            <legend className={legend}>RAM</legend>
            <div className="flex flex-wrap gap-2">
              {rams.map((gb) => (
                <Chip key={gb} on={ram.has(gb)} label={`${gb} GB`} onChange={() => setRam((s) => toggle(s, gb))} />
              ))}
            </div>
          </fieldset>
        )}

        <fieldset className="m-0 border-0 p-0">
          <legend className={legend}>Shop</legend>
          <div className="flex flex-col gap-1">
            {shops.map((shop) => (
              <label key={shop} className={`${checkRow} min-h-9`}>
                <input type="checkbox" checked={shopsOn.has(shop)} onChange={() => setShopsOn((s) => toggle(s, shop))} className={checkbox} />
                {shop}
              </label>
            ))}
          </div>
        </fieldset>

        <label className={`${checkRow} font-semibold`}>
          <input type="checkbox" checked={inStockOnly} onChange={() => setInStockOnly((v) => !v)} className={checkbox} />
          In stock only
        </label>
      </aside>

      <section aria-label="Phones" className="rise min-w-0 flex-[999_1_640px]">
        <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
          <p className="m-0 text-[15px] text-ink-2" aria-live="polite">
            <strong className="text-ink">
              {shown.length} {shown.length === 1 ? 'phone' : 'phones'}
            </strong>
            {summary}
          </p>
          <label className="flex items-center gap-2 text-[14px] text-ink-3">
            Sort
            <select
              value={sort}
              onChange={(e) => setSort(e.target.value as Sort)}
              className="h-10 rounded-xl border border-line-strong bg-white px-3 text-[14px] text-ink"
            >
              <option value="low">Price: low to high</option>
              <option value="high">Price: high to low</option>
              <option value="shops">Most shops</option>
            </select>
          </label>
        </div>

        {shown.length === 0 ? (
          <div className="flex flex-col items-start gap-3 rounded-3xl bg-panel px-6 py-5 text-[15px] text-ink-2">
            No phone matches these filters.
            <button type="button" onClick={reset} className="cursor-pointer font-semibold text-accent">
              Clear filters
            </button>
          </div>
        ) : (
          <ul className="m-0 flex list-none flex-wrap gap-4 p-0">
            {shown.map((p) => (
              <li key={p.row.phoneId} className="flex min-w-0 flex-[1_1_260px]">
                <Link href={p.row.href} className="lift edge flex w-full flex-col gap-2.5 rounded-3xl p-[22px]">
                  <span className="text-[17px] font-semibold">{p.row.model}</span>
                  <span className="flex flex-wrap gap-1.5">
                    {[p.row.ramGb !== null ? `${p.row.ramGb} GB` : null, sizeLabel(p.row.storageGb), p.row.esimOnly ? 'eSIM' : null]
                      .filter((c): c is string => c !== null)
                      .map((c) => (
                        <span key={c} className="rounded-lg bg-panel px-[9px] py-1 text-[13px] font-semibold">
                          {c}
```

```ts
import { configLabel, sriLankaDate, type PhoneSummary, type PricePoint } from '@/lib/prices'

/**
 * Phones selling below their usual price. PCPrice.lk's rule, unchanged.
 *
 * A price drop is a phone whose cheapest in-stock price now (any warranty) is
 * at least 5% under the median of its daily lowest in-stock price over the
 * previous 30 days, with at least 7 of those days on record, so a phone we only
 * just started tracking cannot show a drop it never had. Drops over 30% are
 * left out as likely matching errors.
 *
 * "Below usual", not "price cut": it can also be a shop that sells cheaper than
 * the ones we saw before. Either way it is cheaper to buy now.
 */

const THRESHOLD = 0.05
/** Above this the drop is more likely a listing matched to the wrong phone than a real discount. */
const IMPLAUSIBLE = 0.3
const WINDOW_DAYS = 30
const MIN_DAYS = 7

export type Deal = {
  phone: PhoneSummary
  priceLkr: number
  shop: string
  usualLkr: number
  savingLkr: number
  /** Whole percent, rounded. */
  percent: number
  /** Cheaper now than on any day in the window. */
  lowest: boolean
}

function median(values: number[]): number {
  const s = [...values].sort((a, b) => a - b)
  const mid = s.length >> 1
  return s.length % 2 ? s[mid] : (s[mid - 1] + s[mid]) / 2
}

export function findDeals(
  directory: Record<string, PhoneSummary[]>,
  lows: Record<string, PricePoint[]>,
  now = new Date(),
): Deal[] {
  // Sri Lanka days, matching price_history.recorded_on.
  const today = sriLankaDate(now)
  const from = sriLankaDate(new Date(now.getTime() - WINDOW_DAYS * 86_400_000))

  const deals: Deal[] = []
  for (const phones of Object.values(directory)) {
    for (const phone of phones) {
      const best = phone.overall
      if (!best || !best.inStock) continue

      // Today is left out: it is the price being judged, not the norm.
      const inWindow = (lows[phone.phoneId] ?? []).filter((p) => p.day >= from && p.day < today).map((p) => p.lowestLkr)
      if (inWindow.length < MIN_DAYS) continue

      const usual = median(inWindow)
      if (best.priceLkr > usual * (1 - THRESHOLD)) continue
      if (best.priceLkr < usual * (1 - IMPLAUSIBLE)) continue

      deals.push({
        phone,
        priceLkr: best.priceLkr,
        shop: best.shop,
        usualLkr: Math.round(usual),
        savingLkr: Math.round(usual - best.priceLkr),
        percent: Math.round((100 * (usual - best.priceLkr)) / usual),
        lowest: best.priceLkr < Math.min(...inWindow),
      })
    }
  }
  return deals.sort((a, b) => b.percent - a.percent || b.savingLkr - a.savingLkr)
}

/** What the lists and cards render, without the whole phone. */
export type DealItem = {
  brandSlug: string
  brand: string
  phoneId: string
  model: string
  config: string
  priceLkr: number
  shop: string
  usualLkr: number
  savingLkr: number
  percent: number
  lowest: boolean
}

export function toDealItems(deals: Deal[]): DealItem[] {
  return deals.map((d) => ({
    brandSlug: d.phone.brandSlug,
    brand: d.phone.brand,
    phoneId: d.phone.phoneId,
    model: d.phone.model,
    config: configLabel(d.phone),
    priceLkr: d.priceLkr,
    shop: d.shop,
    usualLkr: d.usualLkr,
    savingLkr: d.savingLkr,
    percent: d.percent,
    lowest: d.lowest,
  }))
}
```

```ts
import { configLabel, type PhoneSummary } from '@/lib/prices'

/**
 * Phone search over the directory the pages already load. ~1,300 phones is
 * small enough that a scan beats running a search index.
 *
 * Matching is by word, not substring (PCPrice.lk's lesson): each query word
 * must *start* a word of the name, and a short number must *be* one, so "7"
 * finds the 7 of "Galaxy S7", never the start of "75". Runs of two or three
 * words are also tried joined, so "s26ultra" and "256gb" still match. A name
 * containing the whole query as typed ranks first.
 */
const words = (s: string) => s.toLowerCase().split(/[^a-z0-9]+/).filter(Boolean)
const squash = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, '')

function matchesWord(nameWords: string[], q: string): boolean {
  const exact = /^\d{1,2}$/.test(q)
  for (let i = 0; i < nameWords.length; i++) {
    let joined = ''
    for (let len = 0; len < 3 && i + len < nameWords.length; len++) {
      joined += nameWords[i + len]
      if (exact ? joined === q : joined.startsWith(q)) return true
    }
  }
  return false
}

export function searchPhones(directory: Record<string, PhoneSummary[]>, query: string): PhoneSummary[] {
  const queryWords = words(query)
  if (queryWords.length === 0) return []
  const phrase = squash(query)

  const hits: { phone: PhoneSummary; phrase: boolean }[] = []
  for (const phones of Object.values(directory)) {
    for (const phone of phones) {
      const nameWords = words(`${phone.brand} ${phone.model} ${configLabel(phone)}`)
      if (queryWords.every((q) => matchesWord(nameWords, q))) {
        hits.push({ phone, phrase: squash(`${phone.brand} ${phone.model}`).includes(phrase) || squash(phone.model).includes(phrase) })
      }
    }
  }

  // Then buyable first, then cheapest: the question being asked is where to buy.
  const price = (p: PhoneSummary) => p.overall?.priceLkr ?? Number.MAX_SAFE_INTEGER
  return hits
    .sort(
      (a, b) =>
        Number(b.phrase) - Number(a.phrase) ||
        Number(b.phone.inStockShops > 0) - Number(a.phone.inStockShops > 0) ||
        price(a.phone) - price(b.phone),
    )
    .map((h) => h.phone)
}
```

```ts
/**
 * The last 31 days of each phone's daily lowest in-stock price (any warranty)
 * for one brand: what price drops are judged on. Cached per brand like the
 * directory; 31 days of one number per phone stays far under the entry limit.
 */
export async function readLows(brandSlug: string): Promise<Record<string, PricePoint[]>> {
  const brand = brandFromSlug(brandSlug)
  if (!brand) return {}

  const rows = await getDb()
    .select({
      phoneId: priceHistory.phoneId,
      day: priceHistory.recordedOn,
      lowest: sql<string>`min(${priceHistory.priceLkr})`,
    })
    .from(priceHistory)
    .innerJoin(phones, eq(phones.phoneId, priceHistory.phoneId))
    .where(and(eq(phones.brand, brand), eq(priceHistory.inStock, true), sql`${priceHistory.recordedOn} >= current_date - 31`))
    .groupBy(priceHistory.phoneId, priceHistory.recordedOn)
    .orderBy(asc(priceHistory.phoneId), asc(priceHistory.recordedOn))

  const out: Record<string, PricePoint[]> = {}
  for (const r of rows) (out[r.phoneId] ??= []).push({ day: String(r.day), lowestLkr: Number(r.lowest) })
  return out
}

export const loadLows = unstable_cache(readLows, ['brand-lows'], CACHE)

/** Today's price drops across every brand. Computed from cached reads; costs no extra query. */
export async function loadDeals(): Promise<Deal[]> {
  const [directory, lows] = await Promise.all([loadDirectory(), Promise.all(BRAND_SLUGS.map((s) => loadLows(s)))])
  return findDeals(directory, Object.assign({}, ...lows))
}
```

```ts
import type { MetadataRoute } from 'next'

import type { PhoneSummary } from '@/lib/prices'
import { canonical } from '@/lib/site'
import { loadDeals, loadDirectory } from '@/queries/phones'

/**
 * Every indexable URL: home, price drops (only while there are any), each
 * brand, each phone. `lastModified` is the day a price was last confirmed,
 * not the build time, so a crawler can come back to the pages that moved.
 */
export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const [directory, deals] = await Promise.all([loadDirectory(), loadDeals()])
  const brands = Object.entries(directory).filter(([, list]) => list.length > 0)
  const newest = (list: PhoneSummary[]) => list.map((p) => p.updatedOn).sort().at(-1)!

  return [
    { url: canonical('/'), lastModified: new Date(), changeFrequency: 'daily', priority: 1 },
    ...(deals.length > 0
      ? [{ url: canonical('/deals'), lastModified: new Date(), changeFrequency: 'daily' as const, priority: 0.8 }]
      : []),
    ...brands.map(([slug, list]) => ({
      url: canonical(`/${slug}`),
      lastModified: new Date(newest(list)),
      changeFrequency: 'daily' as const,
      priority: 0.8,
    })),
    ...brands.flatMap(([slug, list]) =>
      list.map((p) => ({
        url: canonical(`/${slug}/${p.phoneId}`),
        lastModified: new Date(p.updatedOn),
        changeFrequency: 'daily' as const,
        priority: 0.6,
      })),
    ),
  ]
}
```

```ts
/** Cloudflare's Turnstile page, which GSMArena puts in front of its search. We never get past it. */
export function isChallenge(html: string): boolean {
  return /challenges\.cloudflare\.com|Turnstile check/i.test(html)
}

const text = (s: string) => s.replace(/\s+/g, ' ').trim()

/** A brand list page: each phone's name and path, newest first, and how many pages the list has. */
export function parseMakerPage(html: string): { title: string; candidates: Candidate[]; lastPage: number } {
  const $ = cheerio.load(html)
  const candidates = $('#list-brands .general-menu li a')
    .toArray()
    .flatMap((a) => {
      const path = $(a).attr('href')
      const name = text($(a).find('strong').text())
      return path && name ? [{ name, path }] : []
    })
  const last = $('#last-button').attr('href')?.match(/-p(\d+)\.php$/)
  return { title: text($('title').first().text()), candidates, lastPage: last ? Number(last[1]) : 1 }
}

export type ParsedSpecs = {
  name: string
  /** The page's canonical (www) URL, for the attribution link. */
  url: string | null
  mainCamera: string | null
  display: string | null
  network: string | null
  colours: string | null
}

/** A spec page: the `data-spec` cells mobile-specs-api reads, reduced to what the phone page shows. */
export function parseSpecPage(html: string): ParsedSpecs {
  const $ = cheerio.load(html)
  const spec = (key: string) => text($(`[data-spec="${key}"]`).first().text()) || null

  const camera = spec('cam1modules')?.match(/(\d+(?:\.\d+)?)\s*MP/i)
  const size = spec('displaysize')?.match(/^(\d+(?:\.\d+)?)\s*inches/i)
  // "Dynamic LTPO AMOLED 2X, 120Hz, HDR10+, 2600 nits (peak)" → the panel and its refresh rate.
  const type = spec('displaytype')?.split(',').slice(0, 2).map((s) => s.trim()).join(', ') ?? null
  const display = [size ? `${size[1]}-inch` : null, type].filter(Boolean).join(' ') || null
  const tech = spec('nettech')
  const network = !tech
    ? null
    : /\b5G\b/i.test(tech)
      ? '5G'
      : /\bLTE\b/i.test(tech)
        ? '4G'
        : /HSPA|UMTS|EVDO|CDMA2000/i.test(tech)
          ? '3G'
          : '2G'

  return {
    name: text($('h1').first().text()),
    url: $('link[rel="canonical"]').attr('href') ?? null,
    mainCamera: camera ? `${camera[1]} MP` : null,
    display,
    network,
    colours: spec('colors'),
  }
}

/** Words that may lead a GSMArena name for a brand and are dropped on both sides (rule 1). */
const LEADS: Record<string, string[]> = { Nokia: ['nokia', 'hmd'] }

function words(name: string, brand: string): string {
  const leads = LEADS[brand] ?? [brand.toLowerCase()]
  return name
    .toLowerCase()
    .replace(/\+/g, ' plus ')
    .replace(/[^a-z0-9]+/g, ' ')
    .replace(/\b[45]g\b/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
    .replace(new RegExp(`^(?:${leads.join('|')}) `), '')
}

const networkOf = (model: string) => model.match(/\b([45])G\b/i)?.[1] ?? null

function candidateNetwork(c: Candidate): string | null {
  const fromPath = c.path.match(/_([45])g(?=[-_+.])/i)?.[1]
  return fromPath ?? networkOf(c.name)
}

/** The one GSMArena entry for this brand + model by the rules above, or null. */
export function pickCandidate(phone: { brand: string; model: string }, candidates: Candidate[]): Candidate | null {
  const want = words(phone.model, phone.brand)
  const net = networkOf(phone.model)
  const same = candidates.filter((c) => words(c.name, phone.brand) === want)
  const fitting = net === '5' ? same.filter((c) => candidateNetwork(c) === '5') : net === '4' ? same.filter((c) => candidateNetwork(c) !== '5') : same
  const exact = net ? fitting.filter((c) => candidateNetwork(c) === net) : []
  const pool = exact.length > 0 ? exact : fitting
  return pool.length === 1 ? pool[0] : null
}
```

```ts
import { sql } from 'drizzle-orm'

import { getDb } from '@/db'
import { phoneSpecs, type NewPhoneSpecsRow } from '@/db/schema'
import { politeGet } from '@/lib/http'
import {
  GSMARENA,
  MAKERS,
  isChallenge,
  makerPageUrl,
  parseMakerPage,
  parseSpecPage,
  pickCandidate,
  type Candidate,
} from '@/specs/gsmarena'

/**
 * The nightly specs job: for each (brand, model) with no stored specs, find its
 * GSMArena page by rules and keep main camera, display, network and colours.
 *
 * Conduct (docs/decisions.md, 2026-10-25): one request every 5 s, no retries,
 * the honest bot user agent from src/lib/http.ts, and the run stops on 403, 429
 * or a bot check; we never try to get past one. GSMArena's search sits behind a
 * bot check, so models are found on the brand's list pages (newest first),
 * read only until every wanted model of that brand is found.
 *
 * Idempotent: a matched model is never fetched again; an unmatched one is
 * retried after 30 days; a re-run after a stop carries on where it left off.
 * Nothing fetched here is ever sent to Claude.
 */

export const GAP_MS = 5000
export const RETRY_AFTER_DAYS = 30

export type SpecsOptions = { maxRequests?: number; dryRun?: boolean; log?: (m: string) => void }

export type SpecsResult = {
  wanted: number
  requests: number
  matched: number
  unmatched: number
  /** GSMArena answered 403, 429 or a bot check: the run stopped and failed. */
  blocked: boolean
  stopped: string | null
}

type Page = { status: number; body: string; url: string; blocked: boolean }

class Stop extends Error {}

const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms))

/** GETs at least GAP_MS apart, counted against a budget. No retries: each would be another request. */
function pacedGetter(maxRequests: number) {
  let last = 0
  let count = 0
  return {
    get count() {
      return count
    },
    async get(url: string): Promise<Page> {
      if (count >= maxRequests) throw new Stop(`request budget of ${maxRequests} used`)
      const wait = last + GAP_MS - Date.now()
      if (wait > 0) await sleep(wait)
      last = Date.now()
      count++
      const res = await politeGet(url, { retries: 0 })
      return {
        status: res.status,
        body: res.body,
        url: res.url,
        blocked: res.status === 403 || res.status === 429 || isChallenge(res.body),
      }
    },
  }
}

export async function runSpecs(opts: SpecsOptions = {}): Promise<SpecsResult> {
  const log = opts.log ?? console.log
  const http = pacedGetter(opts.maxRequests ?? 1000)
  const db = getDb()

  const res = await db.execute(sql`
    select distinct p.brand, p.model
    from phones p
    left join phone_specs s on s.brand = p.brand and s.model = p.model
    where s.brand is null
       or (not s.matched and s.fetched_at < now() - make_interval(days => ${RETRY_AFTER_DAYS}))
    order by p.brand, p.model
  `)
  const wanted = res.rows as { brand: string; model: string }[]
  const byBrand = new Map<string, string[]>()
  for (const w of wanted) byBrand.set(w.brand, [...(byBrand.get(w.brand) ?? []), w.model])

  const result: SpecsResult = { wanted: wanted.length, requests: 0, matched: 0, unmatched: 0, blocked: false, stopped: null }

  const save = async (row: NewPhoneSpecsRow) => {
    if (opts.dryRun) return
    const values = { ...row, fetchedAt: new Date() }
    await db.insert(phoneSpecs).values(values).onConflictDoUpdate({ target: [phoneSpecs.brand, phoneSpecs.model], set: values })
  }

  const fetchPage = async (url: string): Promise<Page> => {
    const page = await http.get(url)
    if (page.blocked) {
      result.blocked = true
      throw new Stop(`HTTP ${page.status}${isChallenge(page.body) ? ' with a bot check' : ''} at ${url}`)
    }
    return page
  }

  try {
    for (const [brand, models] of byBrand) {
      const makers = MAKERS[brand]
      if (!makers) {
        log(`  ${brand}: no GSMArena list configured, ${models.length} models skipped`)
        continue
      }

      // Newest first; stop paging once every wanted model of this brand is found.
      const candidates: Candidate[] = []
      const found = new Map<string, Candidate>()
      let complete = true
      lists: for (const maker of makers) {
        for (let page = 1, lastPage = 1; page <= lastPage; page++) {
          const list = await fetchPage(makerPageUrl(maker, page))
          if (list.status !== 200) {
            log(`  ${brand}: HTTP ${list.status} on list page ${page}`)
            complete = false
            break
          }
          const parsed = parseMakerPage(list.body)
          lastPage = parsed.lastPage
          candidates.push(...parsed.candidates)
          for (const model of models) {
            if (found.has(model)) continue
            const c = pickCandidate({ brand, model }, candidates)
            if (c) found.set(model, c)
          }
          if (found.size === models.length) break lists
        }
      }

      for (const model of models) {
        const c = found.get(model)
        if (!c) {
          // Only after reading every list page: a failed read proves nothing.
          if (complete) {
            await save({ brand, model, matched: false })
            result.unmatched++
            log(`  unmatched: ${brand} ${model}`)
          }
          continue
        }
        const page = await fetchPage(`${GSMARENA}/${c.path}`)
        if (page.status !== 200) {
          log(`  ${brand} ${model}: HTTP ${page.status} on ${c.path}`)
          continue
        }
        const specs = parseSpecPage(page.body)
        await save({
          brand,
          model,
          matched: true,
          gsmarenaUrl: specs.url ?? `https://www.gsmarena.com/${c.path}`,
          mainCamera: specs.mainCamera,
          display: specs.display,
          network: specs.network,
          colours: specs.colours,
        })
        result.matched++
        log(`  matched: ${brand} ${model} -> ${c.path}`)
      }
    }
  } catch (err) {
    if (!(err instanceof Stop)) throw err
    result.stopped = err.message
  }

  result.requests = http.count
  log(`specs: ${result.matched} matched, ${result.unmatched} unmatched, ${result.requests} requests, ${result.wanted} models wanted`)
  if (result.stopped) log(`stopped: ${result.stopped}`)
  return result
}

/**
 * `npm run specs -- --probe`: does GSMArena answer this machine, are the brand
 * list ids right, and do the selectors still find things? About 20 requests,
 * 5 s apart. Prints status codes, counts and yes/no only, never page text.
 */
export async function probeSpecs(log: (m: string) => void = console.log): Promise<boolean> {
  const http = pacedGetter(40)
  let ok = true

  const index = await http.get(`${GSMARENA}/makers.php3`)
  log(`makers.php3: HTTP ${index.status} ${index.url}${index.blocked ? ' BLOCKED' : ''}`)
  if (index.blocked) {
    log('probe: FAILED')
    return false
  }
  const listed = new Map([...index.body.matchAll(/href="([a-z0-9]+)-phones-(\d+)\.php"/g)].map((m) => [m[1], Number(m[2])]))

  let firstPhone: string | null = null
  for (const [brand, makers] of Object.entries(MAKERS)) {
    for (const maker of makers) {
      const id = listed.get(maker.slug)
      if (id !== maker.id) {
        ok = false
        log(`  ${brand}: MAKERS has ${maker.slug}-${maker.id}, GSMArena lists ${id === undefined ? 'no such brand' : `${maker.slug}-${id}`}`)
      }
      const page = await http.get(makerPageUrl(maker, 1))
      const parsed = parseMakerPage(page.body)
      const titleOk = parsed.title.toLowerCase().includes(maker.slug)
      log(`  ${brand} (${maker.slug}-${maker.id}): HTTP ${page.status}${page.blocked ? ' BLOCKED' : ''}, title ${titleOk ? 'ok' : 'WRONG'}, ${parsed.candidates.length} phones, ${parsed.lastPage} pages`)
      if (page.blocked) {
        log('probe: FAILED')
        return false
      }
      if (page.status !== 200 || !titleOk || parsed.candidates.length === 0) ok = false
      firstPhone ??= parsed.candidates[0]?.path ?? null
    }
  }

  if (firstPhone) {
    const page = await http.get(`${GSMARENA}/${firstPhone}`)
    const s = parseSpecPage(page.body)
    const yes = (v: string | null) => (v ? 'yes' : 'NO')
    log(`  spec page: HTTP ${page.status}${page.blocked ? ' BLOCKED' : ''}, camera ${yes(s.mainCamera)}, display ${yes(s.display)}, network ${yes(s.network)}, colours ${yes(s.colours)}`)
    if (page.status !== 200 || page.blocked || !s.mainCamera || !s.display || !s.network || !s.colours) ok = false
  } else {
    ok = false
  }

  log(ok ? 'probe: ok' : 'probe: FAILED')
  return ok
}
```

- [ ] **Step 4: Create `scripts/specs.ts` and the npm script**

```ts
import { probeSpecs, runSpecs } from '@/pipeline/specs'
import { RunInProgressError, detectActor, finishRun, reapStaleRuns, startRun } from '@/pipeline/runs'

/**
 * CLI for the GSMArena specs job.
 *
 *   npm run specs                         every model still missing (budget 1,000 requests)
 *   npm run specs -- --max-requests 400   stop after 400 requests (about 35 minutes)
 *   npm run specs -- --dry-run            fetch and log, write nothing
 *   npm run specs -- --probe              does GSMArena answer this machine? (~20 requests)
 *
 * Wrapped in a pipeline_runs row and lease, like scrape and normalize. A
 * 403/429/bot check stops the run and exits 1, so it shows in the log or the
 * workflow; a used-up budget is a normal stop.
 */
async function main() {
  const args = process.argv.slice(2)
  if (args.includes('--probe')) process.exit((await probeSpecs()) ? 0 : 1)

  const i = args.indexOf('--max-requests')
  const n = i !== -1 ? Number(args[i + 1]) : NaN
  const maxRequests = Number.isFinite(n) && n > 0 ? n : undefined
  const dryRun = args.includes('--dry-run')

  await reapStaleRuns()

  let handle
  try {
    handle = await startRun({ kind: 'specs', actor: detectActor(), dryRun })
  } catch (err) {
    if (err instanceof RunInProgressError) {
      console.error(err.message)
      process.exit(1)
    }
    throw err
  }

  const lines: string[] = []
  const log = (m: string) => {
    lines.push(m)
    console.log(m)
  }

  try {
    const result = await runSpecs({ maxRequests, dryRun, log })
    await finishRun(handle, { ok: !result.blocked, result, logTail: lines.join('\n') })
    if (result.blocked) process.exit(1)
  } catch (err) {
    await finishRun(handle, { ok: false, error: err, logTail: lines.join('\n') })
    throw err
  }
}

main().catch((err) => {
  console.error(err)
  process.exit(1)
})
```

```ts
export type PhoneSpecs = {
  gsmarenaUrl: string
  mainCamera: string | null
  display: string | null
  network: string | null
  colours: string | null
}

/** GSMArena specs for one brand's models, keyed by model. Matched rows only. */
export async function readSpecs(brandSlug: string): Promise<Record<string, PhoneSpecs>> {
  const brand = brandFromSlug(brandSlug)
  if (!brand) return {}
  const rows = await getDb()
    .select()
    .from(phoneSpecs)
    .where(and(eq(phoneSpecs.brand, brand), eq(phoneSpecs.matched, true)))
  return Object.fromEntries(
    rows.flatMap((r) =>
      r.gsmarenaUrl
        ? [[r.model, { gsmarenaUrl: r.gsmarenaUrl, mainCamera: r.mainCamera, display: r.display, network: r.network, colours: r.colours }]]
        : [],
    ),
  )
}

export const loadSpecs = unstable_cache(readSpecs, ['brand-specs'], CACHE)
```
